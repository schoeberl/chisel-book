\begin{frame}{Testing vs.\ Verification}
	\begin{itemize}
		\item Testing: does the design do what \emph{I} meant?
		\begin{itemize}
			\item Written and run by the developer
			\item Small and fast, so it fits inside the edit-run loop
		\end{itemize}
		\pause
		\item Verification: does the design do what the \emph{specification} says?
		\begin{itemize}
			\item A bigger effort, ideally done by someone other than the developer
			\item Formal verification comes later in the course
		\end{itemize}
		\pause
		\item Why test at all?
		\begin{itemize}
			\item Find bugs while the code is still fresh in your head
			\item Find regressions later, when it is not
			\item Tests document what a module promises
			\item A shared test list is how a team agrees on what ``done'' means
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Running Example: FIFO Queue}
	\begin{itemize}
		\item First-In, First-Out: the first item pushed is the first item popped
		\item Sits between a producer and a consumer that do not run in lockstep,
		      for example on either side of a clock domain crossing
		\item Ours is a chain of stages; an item ripples forward one stage per cycle
	\end{itemize}
	\vspace{1em}

	\end{center}
	\pause
	\vspace{1em}
	Six ports: \code{push}, \code{input} and \code{full} on one side, \code{pop}, \code{output} and \code{empty} on the other
\end{frame}

\begin{frame}{Testing Levels}
	\begin{enumerate}
		\item \textbf{Unit tests}: one component in isolation
		\onslide<2->{\begin{itemize} \item FIFO: a single stage \end{itemize}}

		\item \textbf{Integration tests}: components working together
		\onslide<2->{\begin{itemize} \item FIFO: the whole chain \end{itemize}}

		\item \textbf{System tests}: the complete product, driven by use cases
		\onslide<2->{\begin{itemize} \item FIFO: buffering inside the lab's priority list \end{itemize}}

		\item \textbf{Acceptance tests}: run by the client
		\onslide<2->{\begin{itemize} \item The TA checks your lab \end{itemize}}

		\item \textbf{Non-functional tests}: timing, area, power
		\onslide<2->{\begin{itemize} \item Does it still fit the FPGA and meet the clock? \end{itemize}}
	\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Test-Driven Development (TDD)}
	\begin{itemize}
		\item Write the test \textbf{before} the hardware
		\item Short cycle: \textbf{Red} $\rightarrow$ \textbf{Green} $\rightarrow$ \textbf{Refactor}
	\end{itemize}
	\pause
	Take the first item from your list: \emph{initialized to empty}
	\begin{chisel}
"A stage" should "start empty" in {
  test(new FifoStage(8)) { c =>
    c.io.empty.expect(true.B)
    c.io.full.expect(false.B)
  }
}
	\end{chisel}
	\pause
	\begin{enumerate}
		\item Run it. It does not even compile, there is no \code{FifoStage} yet (\textbf{red})
		\item Write just enough hardware: one register, initialised to empty (\textbf{green})
		\item Clean up, run again, take the next item from the list
	\end{enumerate}
\end{frame}

\begin{frame}{Designing for Testability}
	\begin{itemize}
		\item A test can only control the inputs and observe the outputs
		\item From the ports of the FIFO you cannot see what is in D2
		\pause
		\item So how do you test ``items get to the front without popping''?
		\pause
		\begin{itemize}
			\item Wait long enough, then pop and check: slow and indirect
			\item Add a debug port that exposes the stages: costs hardware, only in test builds?
			\item Reuse the stages as one long shift register: a \textbf{scan chain}
		\end{itemize}
		\pause
		\item The general rule: small modules with clear ports are the easy ones to test
		\item Keeping control and data paths apart helps, so does keeping all stages identical
	\end{itemize}
\end{frame}

\begin{frame}{Property-Based Testing (PBT)}
	\begin{itemize}
		\item Your unit tests poke a handful of values that you picked by hand
		\item The input space of even a small FIFO is enormous: which values, how many, in which order?
		\pause
		\item PBT turns it around: state a \textbf{property} that must hold for \emph{all} inputs,
		      and let a tool pick the inputs
		\begin{enumerate}
			\item Name the property: ``push then pop returns the value''
			\item Describe the valid inputs with a \textbf{generator}
			\item Run the property a few hundred times with random inputs
		\end{enumerate}
		\pause
		\item Properties about order need an expected answer.
		      A Scala \code{List} is a fine reference model for a FIFO
		\pause
		\item When a run fails, the input is \textbf{shrunk} to the smallest one that still fails
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Property-Based Testing in Chisel}
	\begin{itemize}
		\item ScalaCheck picks the inputs, ScalaTest runs the property, chiseltest drives the hardware
		\item Property: what goes into a stage comes out of it
	\end{itemize}
	\begin{chisel}[basicstyle=\scriptsize\ttfamily, aboveskip=1mm, belowskip=0mm]
"A stage" should "pop what was pushed" in {
  forAll(Gen.choose(0, 255)) { in =>
    test(new FifoStage(8)) { c =>
      c.io.push.poke(true.B)
      c.io.input.poke(in.U)
      c.clock.step()
      c.io.push.poke(false.B)
      c.io.pop.poke(true.B)
      c.io.output.expect(in.U)
      c.clock.step()
      c.io.empty.expect(true.B)
    }
  }
}
	\end{chisel}
\end{frame}

\begin{frame}[fragile]{Example: Order is Preserved}
	\begin{chisel}[basicstyle=\scriptsize\ttfamily, aboveskip=0mm, belowskip=0mm]
val depth = 8
val items = Gen.resize(depth, Gen.listOf(Gen.choose(0, 255)))
"A FIFO" should "pop items in push order" in {
  forAll(items) { list =>
    test(new Fifo(8, depth)) { c =>
      for (item <- list) {  // push
        while (c.io.full.peek().litToBoolean) c.clock.step()
        c.io.push.poke(true.B)
        c.io.input.poke(item.U)
        c.clock.step()
        c.io.push.poke(false.B)
      }
      for (item <- list) {  // pop
        while (c.io.empty.peek().litToBoolean) c.clock.step()
        c.io.output.expect(item.U)
        c.io.pop.poke(true.B)
        c.clock.step()
        c.io.pop.poke(false.B)
      }
    }
  }
}
	\end{chisel}
\end{frame}

\begin{frame}[fragile]{Generators, \code{Arbitrary} and Shrinking}
	\begin{itemize}
		\item A \code{Gen} builds a random value: \code{choose}, \code{listOf}, \code{oneOf}, combined with \code{map}/\code{flatMap}
		\item \code{Arbitrary[T]}: the default generator for a type, used when \code{forAll} is given no generator
		\item The default \code{Arbitrary[Int]} gives negatives and \code{Int.MaxValue}; \code{.U} dislikes both, so provide your own
	\end{itemize}
	\begin{chisel}[basicstyle=\scriptsize\ttfamily, aboveskip=1mm, belowskip=1mm]
implicit val bytes: Arbitrary[List[Int]] =
  Arbitrary(Gen.resize(8, Gen.listOf(Gen.choose(0, 255))))
	\end{chisel}
	\pause
	\begin{itemize}
		\item \textbf{Shrinking}: after a failure, ScalaCheck retries with smaller inputs and reports the smallest one that still fails
		\item Shrinking ignores your generator's constraints; \code{implicit val noShrink: Shrink[Int] = Shrink.shrinkAny} turns it off
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Getting Started}
	\begin{enumerate}
		\item Add the ScalaTest/ScalaCheck bridge to \code{build.sbt}
		\begin{chisel}[basicstyle=\scriptsize\ttfamily]
libraryDependencies += "org.scalatestplus" %%
  "scalacheck-1-18" % "3.2.19.0" % Test
		\end{chisel}
		\item Import it and mix it into the test class you already have
		\begin{chisel}[basicstyle=\scriptsize\ttfamily]
import org.scalacheck.{Arbitrary, Gen}
import org.scalatestplus.scalacheck._

class FifoSpec extends AnyFlatSpec
    with ChiselScalatestTester
    with ScalaCheckPropertyChecks {
  "A FIFO" should "..." in {
    forAll(items) { list => ... }
  }
}
		\end{chisel}
		\item \code{sbt test}, as always
		\item More or fewer runs: \code{forAll(items, minSuccessful(200))}
	\end{enumerate}
\end{frame}

\begin{frame}{Continuous Integration (CI)}

	\textbf{Tests are only useful if they are run!}

	\vspace{1em}
	\begin{itemize}
		\item ``It works on my machine'' means little: you forgot half the tests, and your Java is a different version anyway
		\pause
		\item A CI server checks out every push and every pull request, in a clean environment, and runs \code{sbt test}
		\item The result is a red or green mark on the pull request; a red one cannot be merged
		\item So \code{master} always builds and always passes
		\pause
		\item The same machinery can build the documentation, run the slow benchmarks nightly, or program an FPGA
		\item Worth it even when you work alone
		\item We use GitHub Actions. Alternative: GitLab CI, Jenkins, CircleCI, Travis CI
	\end{itemize}

\end{frame}

\begin{frame}[fragile]{GitHub Actions in Three Steps}
	\begin{enumerate}
		\item Put this in \code{.github/workflows/ci.yml}
		\begin{chisel}[basicstyle=\scriptsize\ttfamily]
name: CI
on: [push, pull_request]
jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - uses: sbt/setup-sbt@v1
      - run: sbt test
		\end{chisel}
		\item Commit and push. The run shows up under the \emph{Actions} tab within a minute
		\item In the repository settings, require the check to pass before a pull request can be merged
	\end{enumerate}
\end{frame}

\begin{frame}{Lab Session: Priority List}

	\begin{itemize}
		\item A hardware \textbf{priority queue}: push key/value pairs in any order, pop them smallest key first
		\item Built as a \textbf{systolic array}: a chain of identical nodes, each holding one pair
		\pause
		\item \textbf{Push}: the new pair enters at the front and bubbles towards its place, one node per cycle
		\item \textbf{Pop}: the pair with the smallest key leaves from the end
		\item Each node only talks to its two neighbours, so the clock rate does not depend on the length
		\pause
		\item Throughput: one pop every two cycles, after the initial latency
		\item Parameters: key width, value width, and number of nodes
	\end{itemize}

\end{frame}

\begin{frame}{Commands and Configuration}

	\textbf{Commands}
	\begin{itemize}
		\item \texttt{IDLE} -- Do nothing
		\item \texttt{PUSH} -- Insert a key/value pair at the front of the array
		\item \texttt{POP} -- Remove the pair with the smallest key from the end
		\item \texttt{PUSH/POP} -- Both in the same cycle
	\end{itemize}

	\pause
	\vspace{0.5em}
	\textbf{Parameters}
	\begin{itemize}
		\item \texttt{keyWidth} -- Bit width of the key, which decides the order
		\item \texttt{valueWidth} -- Bit width of the value carried along
		\item \texttt{depth} -- Number of nodes in the array
	\end{itemize}

	\vspace{0.5em}
	Each node is its own module, so it can be unit tested on its own

\end{frame}

\begin{frame}{Testing the Priority List}

	\textbf{Exercise: 3 mins.} You have a node module and the full array. What goes on the test list?
	\pause
	\begin{itemize}
		\item \textbf{Unit tests}, one node: holds a pair, compares with its neighbour, forwards the right one, \texttt{IDLE} changes nothing
		\item \textbf{Integration tests}, the array: push then pop returns the pair, pops come out sorted,
		      push and pop in the same cycle, throughput is one pop per two cycles
		\item \textbf{Properties}: every pushed pair is popped exactly once,
		      popped keys never decrease, equal keys come out in push order,
		      \texttt{IDLE} cycles change nothing
		\pause
		\item Which of these need a reference model? A sorted Scala list is enough
	\end{itemize}

\end{frame}

\begin{frame}{Extension: Grouped Values}

	\textbf{New requirements}
	\begin{itemize}
		\item A key can have several values: values are tracked in \textbf{groups}
		\item One \texttt{PUSH} may carry a variable number of values
		\item Values pushed with an existing key \textbf{merge} into that key's group
		\item The maximum group size is a parameter
	\end{itemize}

	\vspace{0.5em}
	\textbf{Extended commands}
	\begin{itemize}
		\item \texttt{Push(0)} -- No operation
		\item \texttt{Pop(0)} -- Pop the lowest key and its whole group
		\item \texttt{Push(x)} -- Push \texttt{x} values with a given key
		\item \texttt{Pop(x)} -- Pop the lowest key group, then push \texttt{x} values with a new key
	\end{itemize}

\end{frame}
